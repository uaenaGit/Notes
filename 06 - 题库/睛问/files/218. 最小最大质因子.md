## 💡 解题思路 
先将 $10 ^ 6$范围内的所有质数记录下来（使用线性筛法），然后将质数表从小枚举到大，若有质因子`primes[j]`，则更新最小和最大质因子。

## 💻 代码实现
> my answer
```cpp
#include <iostream>

using namespace std;

const int N = 1e6 + 10;
bool st[N]; // 存储某数是否被筛掉
int primes[N], cnt;

// 线性筛法生成2~N范围内的质数表
void get_primes() {
    for (int i = 2; i < N; i++) {
        if (!st[i]) primes[cnt++] = i;
        for (int j = 0; primes[j] <= N / i; j++) {
            st[primes[j] * i] = true;
            if (i % primes[j] == 0) break;
        }
    }
}

int main()
{
    int n;
    cin >> n;
    get_primes();
    
    int minFactor = N, maxFactor = 1;
    while (n--) {
        int x;
        cin >> x;
        // 从小到大判断是否有质因子
        
        /*
        由于是从小到大枚举的,
        所以第一个判断 x % primes[j] = 0的primes[j]
        就是该数的最小质因子
        */
        for (int j = 0; j < cnt; j++) {
            bool flag = false; // flag为false说明质因子出现
            if (x > 1 && (x % primes[j] == 0)) {
                flag = true;
                // x /= primes[j]; // 将该质因子除尽
            }
            if (flag) { // 更新质因子
                minFactor = min(minFactor, primes[j]);
                maxFactor = max(maxFactor, primes[j]);
            }
        }
    }
    cout << minFactor << ' ' << maxFactor << endl;

    return 0;
}
```
- **时间复杂度**：`O(n)`
- **空间复杂度**：`O(n)`
