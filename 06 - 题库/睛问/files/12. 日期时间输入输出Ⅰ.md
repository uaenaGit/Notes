> 🔗[12 - 日期时间输入输出](https://sunnywhy.com/camp/3415/model/3442?itemId=2737)

## 📌 题目描述
> （请在此处粘贴题目描述）

按 `YYYY-MM-DD HH:MI:SS` 的格式输入年、月、日、时、分、秒，然后输出它们。

注：

1. 不要使用读入字符串然后分割子串的方式

**输入描述**
满足格式 `YYYY-MM-DD HH:MI:SS` 的字符串（时间范围为 1900-01-012199-12-31）。数据保证一定合法。

**输出描述**
分 6 行按整数形式分别输出年、月、日、时、分、秒，要求高位不满对应位数时用 0 补齐。

**样例**

*输入：*
2020-03-25 08:12:40

*输出：*
2020
03
25
08
12
40

## 💡 解题思路 
使用`scanf`或`cin` 格式化输入。
## 💻 代码实现
> my answer
```c
#include <stdio.h>

int main() {
    int yyyy, mm, dd, hh, mi, ss;
    // 从标准输入读取日期时间，格式为 YYYY-MM-DD HH:MM:SS
    scanf("%d-%d-%d %d:%d:%d", &yyyy, &mm, &dd, &hh, &mi, &ss);
    // 按行打印年月日时分秒
    printf("%d\n%d\n%d\n%d\n%d\n%d", yyyy, mm, dd, hh, mi, ss);
    return 0;
}
```
- **时间复杂度**：`O(1)`
- **空间复杂度**：`O(1)`

> to optimize

使用`cin`，中间字符使用`char c`替换
```cpp
#include <iostream>
using namespace std;

int main() {
    int yyyy, mm, dd, hh, mi, ss;
    char c; // 用于跳过分隔符 '-'、':'
    // 从标准输入读取日期时间
    cin >> yyyy >> c >> mm >> c >> dd >> hh >> c >> mi >> c >> ss;
    // 按行输出年月日时分秒
    cout << yyyy << "\n" << mm << "\n" << dd << "\n" << hh << "\n" << mi << "\n" << ss;
    return 0;
}
```
- **时间复杂度**：`O(1)`
- **空间复杂度**：`O(1)`

> others' answer

使用 `stringstream`流对象。
```cpp
#include <iostream>
#include <string>
#include <sstream>    // 核心头文件：stringstream
#include <vector>
#include <iomanip>    // 用于格式化输出 (setw, setfill)

using namespace std;

struct DateTime {
    int year, month, day;
    int hour, minute, second;
    bool valid; // 标记解析是否成功
};

// 函数：尝试解析一行字符串
DateTime parseLine(const string& line) {
    DateTime dt = {0, 0, 0, 0, 0, 0, false};
    
    // 1. 创建 stringstream 对象，载入整行字符串
    stringstream ss(line);
    
    char dash1, dash2, space, colon1, colon2;

    // 2. 按顺序提取：整数 -> 字符 -> 整数 ...
    // 如果格式不匹配（例如某处不是数字），提取操作会失败，流状态变为 false
    if (ss >> dt.year >> dash1 >> dt.month >> dash2 >> dt.day 
            >> space >> dt.hour >> colon1 >> dt.minute >> colon2 >> dt.second) {
        
        // 3. 关键步骤：验证分隔符是否正确
        // stringstream 提取字符时会跳过空白，但这里的逻辑要求严格匹配位置
        // 注意：>> 运算符提取 char 时会自动跳过前面的空白符，但对于紧挨着数字的符号通常能正确捕获
        // 为了更严谨，我们可以检查捕获到的字符
        if (dash1 == '-' && dash2 == '-' && space == ' ' && colon1 == ':' && colon2 == ':') {
            
            // 4. 额外检查：确保流中没有多余的垃圾字符（可选）
            char extra;
            if (!(ss >> extra)) {
                dt.valid = true; // 没有多余字符，解析完美成功
            } else {
                // 有额外字符，视需求决定是否算成功。这里我们算作成功但给出提示，或者设为无效
                // 为了演示简单，这里依然视为有效，只要前缀匹配
                dt.valid = true; 
            }
        }
    }
    
    return dt;
}

int main() {
    string line;
    cout << "请输入日期时间 (格式: YYYY-MM-DD HH:MM:SS)，按 Ctrl+D/Z 结束:" << endl;

    while (getline(cin, line)) {
        if (line.empty()) continue;

        DateTime dt = parseLine(line);

        if (dt.valid) {
            // 格式化输出
            cout << "解析成功: "
                 << setw(4) << setfill('0') << dt.year << "-"
                 << setw(2) << setfill('0') << dt.month << "-"
                 << setw(2) << setfill('0') << dt.day << " "
                 << setw(2) << setfill('0') << dt.hour << ":"
                 << setw(2) << setfill('0') << dt.minute << ":"
                 << setw(2) << setfill('0') << dt.second << endl;
        } else {
            cout << "格式错误: 无法解析 \"" << line << "\"" << endl;
            cout << "请确保格式为 YYYY-MM-DD HH:MM:SS 且分隔符正确。" << endl;
        }
    }

    return 0;
}
```
- **时间复杂度**：`O(...)`
- **空间复杂度**：`O(...)`
