> 🔗[425 - 最小消耗能量](https://sunnywhy.com/camp/3415/model/3590?itemId=3002)

## 📌 题目描述
> （请在此处粘贴题目描述）

### 题目描述
现有 $n$ 个从左至右摆放着的高台（编号为从 $1$ 到 $n$），每个高台有各自的高度 $h_i$。假设闯关者当前处于第 $i$ 个高台，那么可以选择跳到第 $i+1$ 或第 $i+2$ 个高台（闯关者能够跳任意高度）。如果从第 $i$ 个高台跳到第 $j$ 个高台，那么将会消耗闯关者 $|h_j - h_i|$ 点能量。问从第 $1$ 个高台出发、到达第 $n$ 个高台的过程中需要消耗的最小能量。

### 输入描述
- 第一行一个整数 $n$（$1 \le n \le 10^4$），表示高台个数。
- 第二行为用空格隔开的 $n$ 个整数 $h_i$（$1 \le h_i \le 100$），表示各高台的高度。

### 输出描述
一个整数，表示需要消耗的最小能量。

### 样例1
**输入**
4
2 5 1 3

**输出**
3

**解释**
从第 $1$ 个高台出发，跳到第 $3$ 个高台，消耗 $|1 - 2| = 1$ 点能量；然后跳到第 $4$ 个高台，消耗 $|3 - 1| = 2$ 点能量。此方案消耗的能量最小，为 $1 + 2 = 3$ 点。
## 💡 解题思路 
使用`dfs(i)` 表示跳到第 `i` 个高台消耗的最少能量。
则`dfs(i) = min(dfs(i - 1) + |h[i]| - h[i-1]|, dfs(i - 2) + |h[i] - h[i-2]|)`。

第一版代码：（部分样例不通过）
```cpp
#include <iostream>
#include <cmath>
#include <cstring>

using namespace std;
const int N = 10010;
int h[N];
long long memo[N];

// dfs(i)表示跳到i高台花费的最少能量
long long dfs(int i) {
    if (i <= 1) {
        return 0; // 这里有问题 当 i = 0 时不可能跳到这儿来 应初始化为+∞
        // 造成会有虚假的更小的能量消耗的路径 因为h[0] = 0
    }
    long long &res = memo[i];
    if (res != -1) {
        return res;
    }
    res = min(dfs(i - 1) + abs(h[i] - h[i - 1]), dfs(i - 2) + abs((h[i] - h[i - 2])));
    return res;
}

int main()
{
    int n;
    scanf("%d", &n);
    memset(memo, -1, sizeof(memo));
    for (int i = 1; i <= n; i++) scanf("%d", &h[i]);

    printf("%lld", dfs(n));

    return 0;
}
```
修改后：
```cpp
#include <iostream>
#include <cmath>
#include <cstring>

using namespace std;
const int N = 10010;
int h[N];
long long memo[N];
const long long INF = 1e18; 

// dfs(i)表示跳到i高台花费的最少能量
long long dfs(int i) {
    if (i == 1) {
        return 0;
    }
    if (i < 1) {
        return INF;
    }
    long long &res = memo[i];
    if (res != -1) {
        return res;
    }
    res = min(dfs(i - 1) + abs(h[i] - h[i - 1]), dfs(i - 2) + abs((h[i] - h[i - 2])));
    return res;
}

int main()
{
    int n;
    scanf("%d", &n);
    memset(memo, -1, sizeof(memo));
    for (int i = 1; i <= n; i++) scanf("%d", &h[i]);

    printf("%lld", dfs(n));

    return 0;
}
```

- **时间复杂度**：$O(n)$。
*   **状态总数**：你的 `dfs` 函数只有一个参数 `i`（表示当前所在的台阶编号），`i` 的取值范围是 $1$ 到 $N$。因此，总共有 $N$ 个不同的状态。
*   **单个状态的计算成本**：
    *   对于每个状态 `i`，代码执行了两次递归调用：`dfs(i-1)` 和 `dfs(i-2)`。
    *   由于使用了 `memo` 数组进行记忆化，**每个状态只会被完整计算一次**。
    *   当第二次需要该状态时，直接通过 `if (res != -1) return res;` 在 $O(1)$ 时间内返回。
    *   每次计算包含常数次操作（减法、绝对值 `abs`、加法、`min` 比较）。
*   **结论**：总时间 = 状态数 $\times$ 单个状态计算时间 = $N \times O(1) = O(N)$。

- **空间复杂度**：$O(n)$

```cpp
#include <iostream>
#include <cmath>
#include <cstring>

using namespace std;
const int N = 10010;
int h[N];
long long f[N];
const long long INF = 1e18; 

int main()
{
    int n;
    scanf("%d", &n);
    for (int i = 1; i <= n; i++) scanf("%d", &h[i]);

    f[0] = INF, f[1] = 0;
    for (int i = 2; i <= n; i++) {
        f[i] = min(f[i-1] + abs(h[i] - h[i-1]), f[i-2] + abs(h[i] - h[i-2]));
    }
    cout << f[n] << endl;
    return 0;
}

// 写法二：
#include <iostream>
#include <cmath>
#include <cstring>

using namespace std;
const int N = 10010;
int h[N];
long long f[N];

int main()
{
    int n;
    scanf("%d", &n);
    for (int i = 1; i <= n; i++) scanf("%d", &h[i]);

    // 2 只能由 1跳过来
    f[1] = 0, f[2] = abs(h[2] - h[1]);
    for (int i = 3; i <= n; i++) {
        f[i] = min(f[i-1] + abs(h[i] - h[i-1]), f[i-2] + abs(h[i] - h[i-2]));
    }
    cout << f[n] << endl;
    return 0;
}
```

- **空间复杂度**：$O(N)$
