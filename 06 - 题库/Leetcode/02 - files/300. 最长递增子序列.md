> 🔗[300 - 最长递增子序列](https://leetcode.cn/problems/longest-increasing-subsequence/description/)

## 📌 题目描述
> （请在此处粘贴题目描述）

给你一个整数数组 `nums` ，找到其中最长严格递增子序列的长度。

**子序列** 是由数组派生而来的序列，删除（或不删除）数组中的元素而不改变其余元素的顺序。例如，`[3,6,2,7]` 是数组 `[0,3,1,6,2,2,7]` 的子序列。

 

**示例 1：**

**输入：** nums = [10,9,2,5,3,7,101,18]
**输出：** 4
**解释：** 最长递增子序列是 [2,3,7,101]，因此长度为 4 。

**示例 2：**

**输入：** nums = [0,1,0,3,2,3]
**输出：** 4

**示例 3：**

**输入：** nums = [7,7,7,7,7,7,7]
**输出：** 1

**提示：**

- `1 <= nums.length <= 2500`
- $-10^4 <= nums[i] <= 10^4$

**进阶：**

- 你能将算法的时间复杂度降低到 `O(n log(n))` 吗?

记忆化搜索。
```cpp
class Solution {
public:
    int lengthOfLIS(vector<int>& nums) {
        int n = nums.size();
        vector<int> memo(n, 0); // 可以初始化为0 因为1个数也是递增子序列
        auto dfs = [&](this auto&& dfs, int i) ->int {
            /*
            由于for循环中j不会小于0所以i不会小于0
            if (i < 0) {
                return 0;
            }
            */
            int &res = memo[i];
            if (res > 0) {
                return res;
            }
            for (int j = i - 1; j >= 0; j--) {
                if (nums[j] < nums[i]) {
                    res = max(res, dfs(j));
                }
            }
            res += 1; // 加上nums[i]这一个元素
            return res;
        };

        int ans = 0;
        for (int i = 0; i < n; i++) {
            ans = max(ans, dfs(i));
        }

        return ans;
    }
};
```
- **时间复杂度**：$O(n^2)$。状态个数是 $O(n)$，单个状态的计算时间是 $O(n)$。
- **空间复杂度**：$O(n)$

```cpp
        vector<int> f(n, 0); // 可以初始化为0 因为1个数也是递增子序列

                if (nums[i] > nums[j]) {
                    f[i] = max(f[i], f[j]);
                }
            }
            f[i] += 1; // 加上自身nums[i]这一个数
        }
        int ans = 0;
        for (int i = 0; i < n; i++) {
            ans = max(ans, f[i]);
        }
        return ans;
    }
};
```

贪心 + 二分。
使用 `g[i]` 表示长度为 `i + 1` 的递增子序列的末尾元素的最小值。（固定长度为定值）
算法：在 `g` 上使用二分查找快速找到第一个 $\geq nums[i]$ 的下标 `j`。如果`j`不存在，那么`nums[j]`直接加到 `g`末尾；否则修改 `g[j]` 为 `nums[i]`。
```cpp
class Solution {
public:
    int lengthOfLIS(vector<int>& nums) {
        int n = nums.size();
        vector<int> g;
        for (int x : nums) {
            // 在g中查找第一个大于等于x的元素下标
            auto it = lower_bound(g.begin(), g.end(), x);
            if (it == g.end()) {
                g.push_back(x);
            } else {
                *it = x;
            }
        }
        return g.size();
    }
};
```
- **时间复杂度**：$O(nlogn)$
- **空间复杂度**：$O(n)$

手动实现二分。
```cpp
class Solution {
public:
    int lengthOfLIS(vector<int>& nums) {
        int n = nums.size();
        vector<int> g;
        auto binary_search = [&](int x) ->int {
            int l = 0, r = g.size() - 1;
            while (l <= r) {
                int mid = l + (r - l) / 2;
                if (g[mid] >= x) {
                    r = mid - 1;
                } else {
                    l = mid + 1;
                }
            }
            return l;
        };
        for (int x : nums) {
            // 在g中查找第一个大于等于x的元素下标
            auto it = binary_search(x);
            if (it == g.size()) {
                g.push_back(x);
            } else {
                g[it] = x;
            }
        }
        return g.size();
    }
};
```
