> 🔗[72 - 编辑距离](https://leetcode.cn/problems/edit-distance/description/)

## 📌 题目描述
> （请在此处粘贴题目描述）

给你两个单词 `word1` 和 `word2`， _请返回将 `word1` 转换成 `word2` 所使用的最少操作数_  。

你可以对一个单词进行如下三种操作：

- 插入一个字符
- 删除一个字符
- 替换一个字符

**示例 1：**

**输入：** word1 = "horse", word2 = "ros"
**输出：** 3
**解释：**
horse -> rorse (将 'h' 替换为 'r')
rorse -> rose (删除 'r')
rose -> ros (删除 'e')

**示例 2：**

**输入：** word1 = "intention", word2 = "execution"
**输出：** 5
**解释：**
intention -> inention (删除 't')
inention -> enention (将 'i' 替换为 'e')
enention -> exention (将 'n' 替换为 'x')
exention -> exection (将 'n' 替换为 'c')
exection -> execution (插入 'u')

**提示：**

- `0 <= word1.length, word2.length <= 500`
- `word1` 和 `word2` 由小写英文字母组成

## 💡 解题思路 
思路与[[1143. 最长公共子序列]]类似
$$
dfs(i, j) = \begin{cases} dfs(i - 1, j - 1)  & s[i] = t[j] \\
\min(dfs(i, j - 1)(插入), dfs(i - 1, j)(删除), dfs(i - 1, j - 1)(替换)) + 1 & s[i] \neq t[j] \end{cases}
$$

记忆化搜索。
```cpp
class Solution {
public:
    int minDistance(string word1, string word2) {
        int n = word1.size(), m = word2.size();
        vector<vector<int>> memo(n, vector<int>(m, -1));

        auto dfs = [&](this auto&& dfs, int i, int j) ->int {
            if (i < 0) {
                return j + 1; // 由于原字符串为空 需要把原字符串都插入j+1个字符
                // [0, j] 共j + 1个字符
            }
            if (j < 0) {
                return i + 1; // 由于目标字符串为空，需要把原字符串删除 i + 1个字符
            }
            int& res = memo[i][j];
            if (res != -1) {
                return res;
            }
            if (word1[i] == word2[j]) {
                return res = dfs(i - 1, j - 1);
            } else {
                return res = min(min(dfs(i, j - 1), dfs(i - 1, j)), dfs(i - 1, j - 1)) + 1;
                                    // 插入         // 删除           // 替换
            }
        };
        // dfs(i, j) 表示将word1的前i个字符变成word2的前j个字符需要的最少操作数
        return dfs(n - 1, m - 1);
    }
};
```
- **时间复杂度**：$O(m * n)$
- **空间复杂度**：$O(m * n)$

```cpp
class Solution {
public:
    int minDistance(string word1, string word2) {
        int n = word1.size(), m = word2.size();
        vector<vector<int>> f(n + 1, vector<int>(m + 1, 0));
        // f[0][0] 表示空串到空串的编辑距离
        // 注意都多插入了一个状态 下标 +1
        for (int j = 0; j < m; j++) f[0][j + 1] = j + 1; // 对 if (i < 0) {return j + 1}的翻译
        for (int i = 0; i < n; i++) f[i + 1][0] = i + 1; // 对 if (j < 0) {return i + 1}的翻译

        for (int i = 0; i < n; i++) {
	        // f[i + 1][0] = i + 1; 上面的初始化也可以放到这儿来
            for (int j = 0; j < m; j++) {
                if (word1[i] == word2[j]) {
                    f[i + 1][j + 1] = f[i][j];
                } else {
                    f[i + 1][j + 1] = min({f[i + 1][j], f[i][j + 1], f[i][j]}) + 1;
                }
            }
        }
        return f[n][m];
    }
};
```
- **时间复杂度**：$O(n * m)$
- **空间复杂度**：$O(n * m)$

>✨ tips
- 注意 `f[0][0]`表示空串到空串的编辑距离，应初始化为0。

空间优化。
```cpp
class Solution {
public:
    int minDistance(string word1, string word2) {
        int n = word1.size(), m = word2.size();
        vector<int> f(m + 1, 0);
        // 注意都多插入了一个状态 下标 +1
        for (int j = 0; j < m; j++) f[j + 1] = j + 1; // 对 if (i < 0) {return j + 1}的翻译

        for (int i = 0; i < n; i++) {
            int pre = f[0]; // f[i][0]
            f[0]++; // 对 if(j < 0) return {i + 1} 的翻译
            for (int j = 0; j < m; j++) {
                int tmp = f[j + 1]; // f[i][j + 1]
                if (word1[i] == word2[j]) {
                    f[j + 1] = pre;
                } else {
                    f[j + 1] = min({f[j], f[j + 1], pre}) + 1;
                }
                pre = tmp;
            }
        }
        return f[m];
    }
};
```
- **时间复杂度**：$O(n * m)$
- **空间复杂度**：$O(m)$
