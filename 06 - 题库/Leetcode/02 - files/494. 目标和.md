> 🔗[494 - 目标和](https://leetcode.cn/problems/target-sum/description/)

## 📌 题目描述
> （请在此处粘贴题目描述）

给你一个非负整数数组 `nums` 和一个整数 `target` 。

向数组中的每个整数前添加 `'+'` 或 `'-'` ，然后串联起所有整数，可以构造一个 **表达式** ：

- 例如，`nums = [2, 1]` ，可以在 `2` 之前添加 `'+'` ，在 `1` 之前添加 `'-'` ，然后串联起来得到表达式 `"+2-1"` 。

返回可以通过上述方法构造的、运算结果等于 `target` 的不同 **表达式** 的数目。

**示例 1：**

**输入：** nums = [1,1,1,1,1], target = 3
**输出：** 5
**解释：** 一共有 5 种方法让最终目标和为 3 。
-1 + 1 + 1 + 1 + 1 = 3
+1 - 1 + 1 + 1 + 1 = 3
+1 + 1 - 1 + 1 + 1 = 3
+1 + 1 + 1 - 1 + 1 = 3
+1 + 1 + 1 + 1 - 1 = 3

**示例 2：**

**输入：** nums = [1], target = 1
**输出：** 1

**提示：**

- `1 <= nums.length <= 20`
- `0 <= nums[i] <= 1000`
- `0 <= sum(nums[i]) <= 1000`
- `-1000 <= target <= 1000`

## 💡 解题思路 
介绍 $01$ 背包：见[[2. 01背包问题]]

常见变形：

对于本题：
![image.png](https://cdn.jsdelivr.net/gh/uaenaGit/image-host@main/images20260315210141779.png)
意味着，只要选出的取正号的元素和恰好等于 $p$，或者取负号的元素和恰好等于 $q$，就等价于表达式的结果恰好等于 `target`。所以问题变成从`nums`中选一些数，使得这些数的元素和恰好等于 $p$ （或者$q$）的方案数。
+ 如果 $target \geq 0$，那么 $q\leq p$ ，取 $q = \frac{s - target} {2}$ 作为背包容量更好。（状态数更少）
+ 如果 $target < 0$，那么 $p < q$，取 $p = \frac {s + target} {2} = \frac{s - |target|} {2}$ 作为背包容量更好。

综上所述，取 $\frac{s - |target|} {2}$ 作为 $01$ 背包的容量是最优的。
注意：
- 如果 $s < 0$，由于 $nums[i] \geq 0$，我们无法得到负数，方案数是 0。
- 如果 $s$ 是奇数，由于 $nums[i]$ 都是整数，我们无法得到非整数 $\frac{s}{2}$，方案数是 0.

```cpp
class Solution {
public:
    int findTargetSumWays(vector<int>& nums, int target) {
        int n = nums.size();
        int m = accumulate(nums.begin(), nums.end(), 0) - abs(target);
        if (m < 0 || m % 2) {
            return 0; // 不能是负数和奇数
        }
        m /= 2;
        // 注意和为m 总共可能 m + 1种状态，从0到m
        vector<vector<int>> memo(n, vector<int>(m + 1, -1)); // -1不可能是答案

        auto dfs = [&](this auto&& dfs, int i, int j) ->int {
            if (i < 0) {
                return j == 0; // 如果 j = 0，表明是一种合法方案
            }
            int& res = memo[i][j]; 
            if (res != -1) { // 之前计算过
                return res;
            }
            if (j < nums[i]) {
                return res = dfs(i - 1, j); // 不能选 nums[i]
            }
            
            return res = dfs(i - 1, j) + dfs(i - 1, j - nums[i]); // 不选 + 选
        };

        // 背包容量为 m
        return dfs(n - 1, m);
    }
};
```
- **时间复杂度**：$O(m * n)$。其中 `n` 为 `nums` 的长度，`m` 为 `nums` 的元素和减去 `∣target∣`（1/2 的常系数可以忽略）。由于每个状态只会计算一次，动态规划的时间复杂度 = 状态个数 × 单个状态的计算时间。本题状态个数等于 `O(nm)`，单个状态的计算时间为 `O(1)`，所以动态规划的时间复杂度为 `O(nm)`。
- **空间复杂度**：$O(m * n)$。
注意：
问：为什么不能在 `c=0` 时就提前返回 1？

答：注意本题 `nums` 中有 `0`，这些 `0` 选或不选都可以。提前返回相当于剩下的 `0`全都不选，只找到了一个方案，这会导致算出的结果比正确答案小。反过来说，如果 `nums` 中的元素都是正数，那么可以在 c=0 时返回 1。

$1 : 1$ 翻译成递推。由于涉及到 `dfs(i - 1)`也即`f[i - 1]`，可以往前加一个状态。
```cpp
class Solution {
public:
    int findTargetSumWays(vector<int>& nums, int target) {
        int n = nums.size();
        int m = accumulate(nums.begin(), nums.end(), 0) - abs(target);
        if (m < 0 || m % 2) {
            return 0; // 不能是负数和奇数
        }
        m /= 2;
        // 注意和为m 总共可能 m + 1种状态，从0到m
        vector<vector<int>> f(n + 1, vector<int>(m + 1, 0));
        f[0][0] = 1;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j <= m; j++) { // 可以枚举到m，因为体积可以为m
                if (j < nums[i]) {
                    f[i + 1][j] = f[i][j]; // 只能不选
                } else {
                    f[i + 1][j] = f[i][j] + f[i][j - nums[i]]; // 选 + 不选
                }
            }
        }

        return f[n][m];
    }
};
```
- **时间复杂度**：$O(mn)$
- **空间复杂度**：$O(mn)$

>✨ tips
- 当涉及到`i - 1`和`i - 2`的下标处理时，可以将状态数组多插入`1`个或`2`个状态。

```cpp
class Solution {
public:
    int findTargetSumWays(vector<int>& nums, int target) {
        int n = nums.size();
        int m = accumulate(nums.begin(), nums.end(), 0) - abs(target);
        if (m < 0 || m % 2) {
            return 0; // 不能是负数和奇数
        }
        m /= 2;
        // 注意和为m 总共可能 m + 1种状态，从0到m
        vector<int> f(m + 1, 0);
        f[0] = 1;
        for (int i = 0; i < n; i++) {
            for (int j = m; j >= nums[i]; j--) { // 倒序枚举
                f[j] = f[j] + f[j - nums[i]];
            }
        }

        // m是体积
        return f[m];
    }
};
```
- **时间复杂度**：$O(nm)$
- **空间复杂度**：$O(m)$

```cpp
            if (res != -1) { // 之前计算过
```

```cpp
            /* 由于是至少，这个 if 不断不需要了，为负也行
            if (j < nums[i]) {
                return res = dfs(i - 1, j); // 不能选 nums[i]
            }
            */
            
            return res = dfs(i - 1, j) + dfs(i - 1, j - nums[i]); // 不选 + 选
        };

        // 背包容量为 m
        return dfs(n - 1, m);
    }

class Solution {
public:
    int findTargetSumWays(vector<int>& nums, int target) {
        int n = nums.size();
        int m = accumulate(nums.begin(), nums.end(), 0) - abs(target);
        if (m < 0 || m % 2) {
            return 0; // 不能是负数和奇数
        }
        m /= 2;
        // 注意和为m 总共可能 m + 1种状态，从0到m
        vector<int> f(m + 1, 1);
        // f[0][0~m] = 1
        
        for (int i = 0; i < n; i++) {
            for (int j = m; j >= 0; j--) { // 由于是至少 可以枚举到0
                f[j] = f[j] + f[max(j - nums[i], 0)]; // 把所有小于等于0的状态都记录到f[0]
            }
        }

        // m是体积
        return f[m];
    }
};
```
