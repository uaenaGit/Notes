> 🔗[322 - 零钱兑换](https://leetcode.cn/problems/coin-change/description/)

## 📌 题目描述
> （请在此处粘贴题目描述）

给你一个整数数组 `coins` ，表示不同面额的硬币；以及一个整数 `amount` ，表示总金额。

计算并返回可以凑成总金额所需的 **最少的硬币个数** 。如果没有任何一种硬币组合能组成总金额，返回 `-1` 。

你可以认为每种硬币的数量是无限的。

**示例 1：**

**输入：** coins = `[1, 2, 5]`, amount = `11`
**输出：**`3` 
**解释：** 11 = 5 + 5 + 1

**示例 2：**

**输入：** coins = `[2]`, amount = `3`
**输出：**-1

**示例 3：**

**输入：** coins = [1], amount = 0
**输出：** 0

**提示：**

- `1 <= coins.length <= 12`
- $1 <= coins[i] <= 2^{31} - 1$
- $0 <= amount <= 10^4$

## 💡 解题思路

本题相当于求从前`n - 1`种硬币中，凑出面值恰好为 `amount`的最小硬币数。

## 💻 代码实现
> my answer
```cpp
class Solution {
public:
    int coinChange(vector<int>& coins, int amount) {
        int n = coins.size();
        vector<vector<int>> memo(n, vector<int>(amount + 1, -1));

        auto dfs = [&](this auto&& dfs, int i, int j) ->int {
            if (i < 0) {
                // 当面额为0时不需要任何硬币
                return (j == 0) ? 0 : INT_MAX / 2; // 防止下面 +1 溢出
            }
            int &res = memo[i][j];
            if (res != -1) {
                return res;
            }
            if (j < coins[i]) {
                return res = dfs(i - 1, j);
            }
            // 不选第i种硬币 + 选第i种硬币(每选1个硬币 j - coins[i],硬币数 +1) 
            // 注意由于硬币个数不限所以还是dfs(i, j - coins[i])
            return res = min(dfs(i - 1, j), dfs(i, j - coins[i]) + 1);
        };


        // dfs(i, j)表示前i种硬币中凑出面值为j的最小硬币数
        int ans = dfs(n - 1, amount);
        if (ans < INT_MAX / 2) return ans;
        else return -1;
    }
};
```
- **时间复杂度**：$O(n * amount)$
- **空间复杂度**：$O(n * amount)$

```cpp
class Solution {
public:
    int coinChange(vector<int>& coins, int amount) {
        int n = coins.size();
        // 由于涉及到-1所以多插入一个状态应该是 n + 1维
        vector<vector<int>> f(n + 1, vector<int>(amount + 1, INT_MAX / 2)); // 初始化为正无穷大
        f[0][0] = 0; // 递归的边界

        for (int i = 0; i < n; i++) {
            for (int j = 0; j <= amount; j++) {
                if (j < coins[i]) {
                    f[i + 1][j] = f[i][j];
                } else {
                    f[i + 1][j] = min(f[i][j], f[i + 1][j - coins[i]] + 1);
                }
            }
        }

        int ans = (f[n][amount] < INT_MAX / 2 ? f[n][amount] : -1);
        return ans;
    }
};
```
- **时间复杂度**：$O(n * amount)$
- **空间复杂度**：$O(n * amount)$

空间优化。
为什么这里`j`可以正序遍历而 $01$ 背包是倒序遍历？
因为状态转移方程`f[i + 1][j] = min(f[i][j], f[i + 1][j - coins[i]] + 1);`两者都在第`i + 1`层。
```cpp
class Solution {
public:
    int coinChange(vector<int>& coins, int amount) {
        int n = coins.size();
        // 由于涉及到-1所以多插入一个状态应该是 n + 1维
        vector<int> f(amount + 1, INT_MAX / 2); // 初始化为正无穷大
        f[0] = 0; // 递归的边界

        for (int i = 0; i < n; i++) {
            for (int j = coins[i]; j <= amount; j++) { // 至少j得大于等于nums[i]
                f[j] = min(f[j], f[j - coins[i]] + 1);
            }
        }

        int ans = (f[amount] < INT_MAX / 2 ? f[amount] : -1);
        return ans;
    }
};
```
- **时间复杂度**：$O(n * amount)$
- **空间复杂度**：$O(amoount)$
