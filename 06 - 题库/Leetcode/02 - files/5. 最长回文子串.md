> 🔗[5 - 最长回文子串](https://leetcode.cn/problems/longest-palindromic-substring/description/)

## 📌 题目描述
> （请在此处粘贴题目描述）

给你一个字符串 `s`，找到 `s` 中最长的 回文 子串。

**示例 1：**

**输入：** s = "babad"
**输出：**"bab"
**解释：**"aba" 同样是符合题意的答案。

**示例 2：**

**输入：** s = "cbbd"
**输出：**"bb"

**提示：**

- `1 <= s.length <= 1000`
- `s` 仅由数字和英文字母组成

## 💡 解题思路 
注意：本题与[[516. 最长回文子序列]] 不同。子串一定是连续的，子序列可以不连续（最长回文子序列长度一定大于等于最长回文子串）。

## 💻 代码实现
> my answer

记忆化搜索。
```cpp
class Solution {
public:
    string longestPalindrome(string s) {
        int n = s.size();
        vector<vector<int>> memo(n, vector<int>(n, -1)); // 记忆化数组
        auto dfs = [&](this auto&& dfs, int i, int j) ->int   {
            if (i >= j) {
                return 1; // 空串或单个字符一定是回文串
            }
            int &res = memo[i][j];
            if (res != -1) {
                return res;
            }
            if (s[i] != s[j]) {
                return res = 0; // 不是回文串
            }
            return res = dfs(i + 1, j - 1);
        };

        int ans = 1; //回文串长度至少为1;
        int st = 0; // 记录回文串起点
        // 从长度为2开始枚举
        for (int len = 2; len <= n; len++) {
            for (int i = 0; i + len - 1 < n; i++) {
                int j = i + len - 1;
                if (s[i]== s[j] && dfs(i + 1, j - 1)) {
                    ans = max(ans, len);
                    st = i;
                }
            }
        }
        return s.substr(st, ans);
    }
};
```
- **时间复杂度**：$O(n^2)$。
**分析过程：**
*   **状态总数**：记忆化数组 `memo` 的大小是 $N \times N$，代表所有可能的子串区间 $(i, j)$。总共有 $\frac{N(N+1)}{2}$ 个状态，即 $O(N^2)$ 个状态。
*   **单个状态的计算成本**：
    *   在 `dfs` 函数内部，除了递归调用外，只进行了常数次的比较和赋值操作 ($O(1)$)。
    *   由于使用了记忆化 (`if (res) return res;`)，每个状态 `(i, j)` **最多只会被计算一次**。后续再次访问该状态时，直接 $O(1)$ 返回。
*   **主循环的调用**：
    *   外层循环枚举长度 `len` 从 2 到 $N$。
    *   内层循环枚举起点 `i`。
    *   总循环次数约为 $\frac{N^2}{2}$ 次。
    *   在循环内部，调用 `dfs(i+1, j-1)`。虽然看起来是在循环里调用递归，但由于记忆化的存在，所有的 `dfs` 调用加起来，总共只会触发 $O(N^2)$ 次实际计算。其余的调用都是 $O(1)$ 的查表。
*   **结论**：总操作次数与状态数量成正比，故时间复杂度为 **$O(N^2)$**。

- **空间复杂度**：$O(n^2)$。
**分析过程：**
*   **记忆化数组**：`vector<vector<int>> memo(n, vector<int>(n, false))` 开辟了一个 $N \times N$ 的二维数组。
    *   占用空间：$N^2 \times \text{sizeof(int)}$，即 **$O(N^2)$**。
*   **递归栈空间**：
    *   `dfs` 函数的最大递归深度取决于区间的收缩速度。每次调用 `i` 加 1，`j` 减 1，区间长度减少 2。
    *   最大深度约为 $N/2$。
    *   占用空间：**$O(N)$**。
*   **其他变量**：循环变量等占用 $O(1)$。
*   **结论**：主导项是二维数组，故总空间复杂度为 **$O(N^2)$**。

```cpp
class Solution {
public:
    string longestPalindrome(string s) {
        int n = s.size();
        vector<vector<int>> memo(n, vector<int>(n, -1)); // 记忆化数组
        vector<vector<int>> f(n, vector<int>(n, 0));
        for (int i = 0; i < n; i++) f[i][i] = 1;
        int ans = 1; //回文串长度至少为1;
        int st = 0; // 记录回文串起点
        // 从长度为2开始枚举
        for (int len = 2; len <= n; len++) {
            for (int i = 0; i + len - 1 < n; i++) {
                int j = i + len - 1;
                if (s[i] != s[j]) {
                    f[i][j] = 0;
                } else {
                    // 关键点：当 len=2 时，i+1 > j-1 (例如 i=0, j=1 -> 1 > 0)
                    // 此时内部是空串，视为回文 (1)
                    if (len == 2) {
                        f[i][j] = 1;
                    } else {
                        f[i][j] = f[i + 1][j - 1];
                    }
                }
                if (f[i][j]) {
                    ans = max(ans, len);
                    st = i;
                }
            }
        }

        return s.substr(st, ans);
    }
};
```
- **时间复杂度**：$O(n ^ 2)$
- **空间复杂度**：$O(n ^ 2)$

第一版（有错）：
```cpp
class Solution {
public:
    string longestPalindrome(string s) {
        int n = s.size();
        vector<int> f(n, 0);
        for (int i = 0; i < n; i++) f[i] = 1;
        int ans = 1; //回文串长度至少为1;
        int st = 0; // 记录回文串起点
        // 从长度为2开始枚举
        for (int len = 2; len <= n; len++) {
            for (int i = n - 1; i >= 0; i--) {
                int j = i + len - 1;
                if (j >= n) continue; // 出界
                if (s[i] != s[j]) {
                    f[j] = 0;
                } else {
                    // 关键点：当 len=2 时，i+1 > j-1 (例如 i=0, j=1 -> 1 > 0)
                    // 此时内部是空串，视为回文 (1)
                    if (len == 2) {
                        f[j] = 1;
                    } else {
                        f[j] = f[j - 1];
                    }
                }
                if (f[j]) {
                    ans = max(ans, len);
                    st = i;
                }
            }
        }

        return s.substr(st, ans);
    }
};
```
- 在 `len` 循环模式下，很难直接用 `f[j-1]` 代表 `len-2` 的状态，因为 `f` 数组没有区分“长度”。上一次更新 `f[j-1]` 是在 `len-1` 或 `len-2` 的循环里吗？
- 实际上，如果你按 `len` 枚举，`f[j]` 在 `len` 循环开始前存的是 `len-1` (或者更短) 的信息，这完全乱了。
**结论**：**“先枚举 `len`”这种写法很难直接优化成一维数组**，因为 `f[j]` 的含义随 `len` 变化，无法复用。

正确的版本：先枚举 `i` 在枚举 `j`。
```cpp
#include <iostream>
#include <vector>
#include <string>
#include <algorithm>

using namespace std;

class Solution {
public:
    string longestPalindrome(string s) {
        int n = s.size();
        if (n == 0) return "";

        // f[j] 表示：在当前的 i 循环中，s[i...j] 是否为回文
        // 在进入 i 循环前，f[j] 存储的是 i+1 行的数据 (即 s[i+1...j])
        vector<int> f(n, 0);
        
        // 初始化：所有单字符都是回文 (i==j 的情况)
        for (int i = 0; i < n; i++) f[i] = 1;

        int ans = 1;
        int st = 0;

        // 【关键修改】：外层枚举 i，从后往前 (n-2 -> 0)
        // 这样保证计算 f[j] 时，f 数组里存的是 i+1 行的数据
        for (int i = n - 2; i >= 0; i--) {
            int prev = 0; // 用于保存 f[j-1] 在“上一行”(i+1) 的值，即 dp[i+1][j-1]
            
            // 内层枚举 j，从 i+1 到 n-1
            for (int j = i + 1; j < n; j++) {
                int temp = f[j]; // 暂存 f[j] (即 dp[i+1][j])，供下一轮 j+1 作为 prev 使用

                if (s[i] != s[j]) {
                    f[j] = 0;
                } else {
                    // 如果长度为 2 (j == i + 1)，内部为空，是回文
                    if (j == i + 1) {
                        f[j] = 1;
                    } else {
                        // 否则，依赖 dp[i+1][j-1]，这正是 prev 保存的值
                        f[j] = prev;
                    }
                }

                // 更新全局答案
                if (f[j]) {
                    if (j - i + 1 > ans) {
                        ans = j - i + 1;
                        st = i;
                    }
                }

                // 更新 prev，准备给下一个 j (即 j+1) 使用
                // 下一个 j 需要的 dp[i+1][j] 就是当前的 temp
                prev = temp;
            }
        }

        return s.substr(st, ans);
    }
};
```
- **时间复杂度**：$O(n ^ 2)$
- **空间复杂度**：$O(n)$

>✨ tips
- 当 `f[i][j]`需要由`f[i + 1][j - 1]`（左下角）的状态转移过来，则要使用变量`prev`存储。
